COMMAND DATABASE PACKAGE MANAGE CONSOLE

# Buat migration baru (setelah mengubah model)
Add-Migration NamaMigration

# Terapkan migration ke database
Update-Database

# Kembali ke migration tertentu
Update-Database NamaMigrationSebelumnya

# Kembalikan database ke kosong (hapus semua tabel migration)
Update-Database 0

# Hapus migration terakhir (yang belum di-apply)
Remove-Migration

# Lihat daftar migration
Get-Migration

# Buat script SQL dari migration
Script-Migration

# Hapus database
Drop-Database
